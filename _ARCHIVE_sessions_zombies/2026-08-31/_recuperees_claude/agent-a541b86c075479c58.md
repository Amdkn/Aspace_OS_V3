DECLARE
  v_tenant text;
  v_user_id uuid;
  v_org_id  uuid;
  v_role    text;
  v_claims  jsonb;
BEGIN
  v_user_id := (event ->> 'user_id')::uuid;
  
  -- GUC discriminator: default to 'solaris_saas' (preserves existing behavior)
  BEGIN
    v_tenant := current_setting('app.tenant', true);
  EXCEPTION WHEN undefined_object THEN
    v_tenant := 'solaris_saas';
  END;
  
  IF v_tenant = 'omk_saas' THEN
    -- OMK path: read omk_saas.memberships
    SELECT org_id, role INTO v_org_id, v_role
    FROM omk_saas.memberships
    WHERE user_id = v_user_id
    LIMIT 1;
    
    IF v_org_id IS NULL THEN
      v_claims := jsonb_build_object('org_id', null, 'role', 'authenticated');
    ELSE
      v_claims := jsonb_build_object('org_id', v_org_id, 'role', COALESCE(v_role, 'member'));
    END IF;
    
    event := jsonb_set(event, '{claims,app_metadata}', 
      COALESCE(event -> 'claims' -> 'app_metadata', '{}'::jsonb) || v_claims);
    event := jsonb_set(event, '{claims,org_id}', to_jsonb(v_org_id));
    event := jsonb_set(event, '{claims,role}', to_jsonb(COALESCE(v_role, 'authenticated')));
    
    RETURN event;
  END IF;
  
  -- ELSE: fall through to existing Solaris SaaS logic (UNCHANGED)
  -- [Preserve the original body for solaris_saas path]
  SELECT m.org_id, m.role INTO v_org_id, v_role
  FROM solaris_saas.memberships m
  WHERE m.user_id = v_user_id
  LIMIT 1;
  
  IF v_org_id IS NULL THEN
    -- staff_user path or no membership
    SELECT s.role INTO v_role
    FROM solaris_internal.staff_users s
    WHERE s.user_id = v_user_id
    LIMIT 1;
    v_claims := jsonb_build_object('is_aaas_staff', v_role IS NOT NULL, 'org_role', COALESCE(v_role, 'authenticated'));
  ELSE
    v_claims := jsonb_build_object('is_aaas_staff', false, 'org_role', COALESCE(v_role, 'member'));
  END IF;
  
  event := jsonb_set(event, '{claims,app_metadata}',
    COALESCE(event -> 'claims' -> 'app_metadata', '{}'::jsonb) || v_claims);
  
  RETURN event;
END;
$$;
